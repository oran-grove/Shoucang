#include <core.p4>
#include <v1model.p4>

// ==========================================================
// 1. 标准网络协议头定义
// ==========================================================
header ethernet_t {
    bit<48> dstAddr; bit<48> srcAddr; bit<16> etherType;
}

header ipv4_t {
    bit<4>  version; bit<4>  ihl; bit<8>  diffserv; bit<16> totalLen;
    bit<16> identification; bit<3>  flags; bit<13> fragOffset;
    bit<8>  ttl; bit<8>  protocol; bit<16> hdrChecksum;
    bit<32> srcAddr; bit<32> dstAddr;
}

header tcp_t {
    bit<16> srcPort; bit<16> dstPort; bit<32> seqNo; bit<32> ackNo;
    bit<4>  dataOffset; bit<4>  res; bit<8>  flags; bit<16> window;
    bit<16> checksum; bit<16> urgPtr;
}

header udp_t {
    bit<16> srcPort; bit<16> dstPort; bit<16> length; bit<16> checksum;
}

header payload_chunk_t {
    bit<128> data;
}

struct headers {
    ethernet_t      ethernet;
    ipv4_t          ipv4;
    tcp_t           tcp;
    udp_t           udp;
    payload_chunk_t payload;
}

// ==========================================================
// 2. 控制面上报结构体（通信契约重构）
// ==========================================================
struct first_pkt_digest_t {
    bit<32>  hash_index;  
    bit<32>  srcAddr;      
    bit<32>  dstAddr;
    bit<8>   protocol;
    bit<8>   has_payload; 
    bit<16>  l4_srcPort;
    bit<16>  l4_dstPort;
    bit<128> payload_data; 
} // 256 bits = 32 Bytes 

// 危险上报
struct audit_digest_t {

    bit<32> hash_index;      // 4B -> 哈希槽位索引
    bit<32> current_pkts;    // 4B -> 触发报警时的累计包数
    bit<32> current_bytes;   // 4B -> 触发报警时的累计流量字节数
    bit<32> sum_entropy;     // 4B -> 累加信息熵


    bit<16> max_entropy;     // 2B -> 周期内最大熵
    bit<16> min_entropy;     // 2B -> 周期内最小熵
    bit<16> trigger_reason;  // 2B -> 触发原因：1=分数超标, 2=流量超标, 3=包数超标


    bit<48> timestamp;       // 6B -> 48位全局硬件时间戳
} // 总长：4 + 4 + 4 + 4 + 2 + 2 + 2 + 6 = 28 字节 (224 bits)

struct metadata {
    bit<16> unified_src_port;
    bit<16> unified_dst_port;
    bit<1>  is_internal_src;  
    bit<1>  is_whitelisted;   
    bit<1>  is_udp;           

    bit<2>  digest_trigger; // 串行分发标记：0=不上报, 1=发首包, 2=发高危报警包
    first_pkt_digest_t msg1_cache; 
    audit_digest_t     msg2_cache; 
}

// ==========================================================
// 3. 解析器 (Parser)
// ==========================================================
parser MyParser(packet_in packet, out headers hdr, inout metadata meta, inout standard_metadata_t standard_metadata) {
    state start { transition parse_ethernet; }

    state parse_ethernet {
        packet.extract(hdr.ethernet);
        transition select(hdr.ethernet.etherType) {
            0x0800: parse_ipv4;
            default: accept;
        }
    }

    state parse_ipv4 {
        packet.extract(hdr.ipv4);
        transition select(hdr.ipv4.protocol) {
            6: parse_tcp;   
            17: parse_udp;  
            default: accept;
        }
    }

    state parse_tcp {
        packet.extract(hdr.tcp);
        transition select(hdr.ipv4.totalLen) {
            56 .. 0xFFFF : parse_payload;
            default : accept;
        }
    }

    state parse_udp {
        packet.extract(hdr.udp);
        transition select(hdr.ipv4.totalLen) {
            44 .. 0xFFFF : parse_payload;
            default : accept;
        }
    }

    state parse_payload {
        packet.extract(hdr.payload);
        transition accept;
    }
}

// ==========================================================
// 4. 核心安检流水线 (Ingress)
// ==========================================================
control MyIngress(inout headers hdr, inout metadata meta, inout standard_metadata_t standard_metadata) {

    // 100万槽位的全局状态表
    register<bit<64>>(1048576) reg_volume_score; 
    register<bit<64>>(1048576) reg_entropy_stat; 

    action drop() { mark_to_drop(standard_metadata); }
    action set_whitelisted() { meta.is_whitelisted = 1; }
    action set_egress(bit<9> port) { standard_metadata.egress_spec = port; }

    // 提权到全局的非参上报 Action
    action send_digest_1() { digest(1, meta.msg1_cache); }
    action send_digest_2() { digest(2, meta.msg2_cache); }

    action update_score_and_audit() {
        // --- 变量声明区 ---
        bit<20> hash_idx; bit<64> val_vol; bit<64> val_ent;
        bit<24> TEN_MB = 10485760; // 10MB 强力防御阈值
        bit<24> SCORE_MAX = 50000;
        bit<16> PKTS_MAX = 6500;    
        bit<16> pkts; bit<24> bytes; bit<24> score;
        bit<12> max_e; bit<12> min_e; bit<40> sum_e; 
        bit<12> cur_entropy = 0;
        
        bit<8> b0; bit<8> b1; bit<8> b2; bit<8> b3; bit<8> b4; bit<8> b5; bit<8> b6; bit<8> b7;
        bit<8> b8; bit<8> b9; bit<8> b10; bit<8> b11; bit<8> b12; bit<8> b13; bit<8> b14; bit<8> b15;
        
        bit<16> new_pkts; bit<24> new_bytes; bit<12> new_max;  bit<12> new_min; bit<40> new_sum;
        bit<24> risk_multi = 1; bit<24> new_score;

        // --- 1. 五元组经典哈希定位 ---
        hash(hash_idx, HashAlgorithm.crc16, (bit<32>)0, { 
            hdr.ipv4.srcAddr, hdr.ipv4.dstAddr, meta.unified_src_port, meta.unified_dst_port, hdr.ipv4.protocol 
        }, (bit<32>)1048576);

        // --- 2. 状态原子读取与高内聚切片 ---
        reg_volume_score.read(val_vol, (bit<32>)hash_idx);
        reg_entropy_stat.read(val_ent, (bit<32>)hash_idx);

        pkts  = val_vol[63:48]; bytes = val_vol[47:24]; score = val_vol[23:0];
        max_e = val_ent[63:52]; min_e = val_ent[51:40]; sum_e = val_ent[39:0]; 

        // --- 3. 实时滑窗邻项异或特征熵计算 ---
        if (hdr.payload.isValid()) {
            b0 = hdr.payload.data[127:120]; b1 = hdr.payload.data[119:112]; b2 = hdr.payload.data[111:104]; b3 = hdr.payload.data[103:96];
            b4 = hdr.payload.data[95:88];   b5 = hdr.payload.data[87:80];   b6 = hdr.payload.data[79:72];   b7 = hdr.payload.data[71:64];
            b8 = hdr.payload.data[63:56];   b9 = hdr.payload.data[55:48];   b10= hdr.payload.data[47:40];   b11= hdr.payload.data[39:32];
            b12= hdr.payload.data[31:24];   b13= hdr.payload.data[23:16];   b14= hdr.payload.data[15:8];    b15= hdr.payload.data[7:0];
            
            cur_entropy = (bit<12>)(b0^b1) + (bit<12>)(b1^b2) + (bit<12>)(b2^b3) + (bit<12>)(b3^b4) + 
                          (bit<12>)(b4^b5) + (bit<12>)(b5^b6) + (bit<12>)(b6^b7) + (bit<12>)(b7^b8) + 
                          (bit<12>)(b8^b9) + (bit<12>)(b9^b10)+ (bit<12>)(b10^b11)+ (bit<12>)(b11^b12)+ 
                          (bit<12>)(b12^b13)+ (bit<12>)(b13^b14)+ (bit<12>)(b14^b15);
        }

        // --- 4. 特征增量更新与评估 ---
        new_pkts  = pkts + 1;
        new_bytes = bytes + (bit<24>)standard_metadata.packet_length;
        new_max   = (cur_entropy > max_e) ? cur_entropy : max_e;
        new_min   = (pkts == 0) ? cur_entropy : ((cur_entropy < min_e) ? cur_entropy : min_e);
        new_sum   = sum_e + (bit<40>)cur_entropy;

        if (cur_entropy > 2500 || cur_entropy < 500) { risk_multi = 4; }
        new_score = score + ((bit<24>)standard_metadata.packet_length * risk_multi);

        // --- 5. 状态机动态决策判定 ---
if (pkts == 0) {
            // 触发首包登记
            meta.digest_trigger = 1; 
            meta.msg1_cache.hash_index = (bit<32>)hash_idx; // 最高字节为 0x00
            meta.msg1_cache.srcAddr = hdr.ipv4.srcAddr; 
            meta.msg1_cache.dstAddr = hdr.ipv4.dstAddr;
            meta.msg1_cache.protocol = hdr.ipv4.protocol; 
            meta.msg1_cache.l4_srcPort = meta.unified_src_port; 
            meta.msg1_cache.l4_dstPort = meta.unified_dst_port;
            meta.msg1_cache.has_payload = hdr.payload.isValid() ? (bit<8>)1 : (bit<8>)0;
            meta.msg1_cache.payload_data = hdr.payload.isValid() ? hdr.payload.data : (bit<128>)0;
        } 
        else if (new_score >= SCORE_MAX || (bit<24>)new_bytes >= TEN_MB || new_pkts >= PKTS_MAX) {
            // 触发 28 字节高危报警
            meta.digest_trigger = 2; 
            
            // 🌟【硬核染色】：将 20 位的 hash_idx 注入低位，最高字节强行染成 0xFF000000 

            meta.msg2_cache.hash_index = ((bit<32>)hash_idx) | 0xFF000000; 
            
            meta.msg2_cache.current_pkts = (bit<32>)new_pkts;
            meta.msg2_cache.current_bytes = (bit<32>)new_bytes;
            meta.msg2_cache.sum_entropy = (bit<32>)new_sum; 
            meta.msg2_cache.max_entropy = (bit<16>)new_max; 
            meta.msg2_cache.min_entropy = (bit<16>)new_min; 

            meta.msg2_cache.timestamp = standard_metadata.ingress_global_timestamp; 
            
            if (new_score >= SCORE_MAX) { meta.msg2_cache.trigger_reason = 1; } 
            else if ((bit<24>)new_bytes >= TEN_MB) { meta.msg2_cache.trigger_reason = 2; } 
            else { meta.msg2_cache.trigger_reason = 3; }

            new_pkts = 0; new_bytes = 0; new_score = 0; new_sum = 0;
            new_max = cur_entropy; new_min = cur_entropy; 
        }
        
        // --- 6. 状态安全复盘落盘 ---
        // 1. 组装 Volume & Score 寄存器（16位 + 24位 + 24位 = 64位）
bit<64> final_vol = ((bit<64>)new_pkts << 48) | 
                     ((bit<64>)new_bytes << 24) | 
                     ((bit<64>)new_score);
reg_volume_score.write((bit<32>)hash_idx, final_vol);

// 2. 组装 Entropy 寄存器（12位最大熵 + 12位最小熵 + 40位累加熵 = 64位）
bit<64> final_ent = ((bit<64>)new_max << 52) | 
                     ((bit<64>)new_min << 40) | 
                     ((bit<64>)new_sum);
reg_entropy_stat.write((bit<32>)hash_idx, final_ent);
    }

    // 表结构声明
    table port_mapping_table {
        key = { standard_metadata.ingress_port : exact; }
        actions = { set_egress; NoAction; }
        size = 256; default_action = NoAction();
    }
    table audit_table {
        actions = { update_score_and_audit; }
        default_action = update_score_and_audit();
    }
    table blacklist_table {
        key = { hdr.ipv4.srcAddr : exact; }
        actions = { drop; NoAction; }
        size = 1024; default_action = NoAction();
    }
    table whitelist_table {
        key = { hdr.ipv4.dstAddr : lpm; }
        actions = { set_whitelisted; NoAction; }
        default_action = NoAction();
    }

    // --- Apply 主控制流块 ---
    apply {
        meta.is_internal_src = 0; meta.is_whitelisted = 0; meta.is_udp = 0;
        meta.unified_src_port = 0; meta.unified_dst_port = 0;
        meta.digest_trigger = 0; 

        if (!port_mapping_table.apply().hit) { drop(); return; }

        if (hdr.ipv4.isValid()) {
            // 内网网段安检
            if ((hdr.ipv4.srcAddr & 0xFF000000) == 0x0A000000 || 
                (hdr.ipv4.srcAddr & 0xFFF00000) == 0xAC100000 || 
                (hdr.ipv4.srcAddr & 0xFFFF0000) == 0xC0A80000) {
                meta.is_internal_src = 1;
            }
            
            // 流向判断
            if (meta.is_internal_src == 1) {
                if (!blacklist_table.apply().hit) {
                    whitelist_table.apply();
                    if (meta.is_whitelisted == 0) {
                        if (hdr.tcp.isValid()) {
                            meta.unified_src_port = hdr.tcp.srcPort; meta.unified_dst_port = hdr.tcp.dstPort;
                            audit_table.apply(); 
                        } else if (hdr.udp.isValid()) {
                            meta.unified_src_port = hdr.udp.srcPort; meta.unified_dst_port = hdr.udp.dstPort;
                            meta.is_udp = 1;
                            audit_table.apply(); 
                        }
                    }
                }
            }
        }

        // 串行流水线末端统一调度发包
        if (meta.digest_trigger == 1) {
            send_digest_1();
        } else if (meta.digest_trigger == 2) {
            send_digest_2();
        }
    }
}

control MyVerifyChecksum(inout headers hdr, inout metadata meta) { apply { } }
control MyComputeChecksum(inout headers hdr, inout metadata meta) { apply { } }
control MyEgress(inout headers hdr, inout metadata meta, inout standard_metadata_t standard_metadata) { apply { } }

control MyDeparser(packet_out packet, in headers hdr) {
    apply {
        packet.emit(hdr.ethernet); packet.emit(hdr.ipv4);
        packet.emit(hdr.tcp); packet.emit(hdr.udp);
        packet.emit(hdr.payload); 
    }
}

V1Switch(MyParser(), MyVerifyChecksum(), MyIngress(), MyEgress(), MyComputeChecksum(), MyDeparser()) main;
